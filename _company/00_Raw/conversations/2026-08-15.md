import React, { useMemo } from 'react';

interface RiskData {
  riskScore: number;
  status: 'Red' | 'Gold' | 'Green';
  error?: string;
}

interface RiskGaugeProps {
  data: RiskData;
  error?: string;
}

// 안전 기본값 정의 (Safe Default Value)
const SAFE_DEFAULT = 'Gold'; 

/**
 * 위험도 게이지 컴포넌트. 데이터에 기반하여 상태를 시각화하고 오류 처리를 수행합니다.
 * @param {RiskData} data - 위험 점수 및 현재 상태 정보
 * @param {string} [error] - 발생한 에러 메시지 (안전 기본값 적용을 위해 사용)
 */
const RiskGaugeComponent: React.FC<RiskGaugeProps> = ({ data, error }) => {
  // 1. 안전 기본값 로직: 에러가 있거나 데이터가 불완전하면 Gold로 설정
  const calculatedStatus = useMemo(() => {
    if (error) {
      // API 실패 시, 안전 기본값을 Red로 설정하여 최고 경고 상태를 유도
      return 'Red'; 
    }
    // 실제 위험 점수 기반 로직 (예시)
    if (data.riskScore >= 85) {
      return 'Red';
    } else if (data.riskScore >= 65) {
      return 'Gold';
    } else {
      return 'Green';
    }
  }, [data.riskScore, error]);

  // 안전 기본값 적용 로직 강화: 데이터가 없거나 유효하지 않을 경우 Gold로 설정
  const finalStatus = error ? 'Red' : calculatedStatus;


  // 시각적 스타일 결정 (CSS 클래스 매핑)
  const statusClass = `status-${finalStatus.toLowerCase()}`;

  return (
    <div className={`risk-gauge ${statusClass}`}>
      <div className="risk-level-text">
        Risk Level: {finalStatus}
      </div>
      {error && (
        <p className="error-message" role="alert">
          Error: Safe Default Applied - {error}
        </p>
      )}
    </div>
  );
};

export default RiskGaugeComponent;
</replace>

### B. 애니메이션 전환 시퀀스 (JS/CSS Transition Logic)

**목표:** 리스크가 Gold에서 Green으로 전환될 때, '통제권 확보'의 성공을 강조하는 미묘한 피드백을 제공합니다.

1.  **Red $\rightarrow$ Gold (위험 인식 및 주의):**
    *   **목표:** 긴급성 부여.
    *   **실행:** `transition` 속도를 빠르게 설정하여 불안정함을 즉각적으로 인지시킵니다.
    *   **Pseudo Code:** `if (currentStatus === 'Red') { transitionDuration = 300ms; }`

2.  **Gold $\rightarrow$ Green (통제권 확보의 순간):**
    *   **목표:** 성공적인 통제권 회복(Control Regained) 시각화. 이 전환이 가장 중요합니다.
    *   **실행:** 전환 중 `var(--color-control)`을 강조하며, **미세한 '안정화' 효과**를 추가합니다.
    *   **Pseudo Code:**
        ```javascript
        // Gold 상태에서 Green으로 변경될 때 (transition 0.5초)
        if (fromStatus === 'Gold' && toStatus === 'Green') {
            // 임시적으로 경고 색상(Gold)을 잠시 유지하며 안정화 효과를 부여
            element.style.setProperty('--gauge-color', '#388E3C'); // Green으로 부드럽게 전환 시작
            element.classList.add('transition-control-secure');
        } else {
            // 일반적인 색상 전환
            element.style.setProperty('--gauge-color', toStatus === 'Green' ? 'var(--color-safe)' : toStatus === 'Red' ? 'var(--color-danger)' : 'var(--color-control)');
        }
        ```

3.  **Green 상태 유지:**
    *   **목표:** 통제권 확보 완료 및 안정성 강조.
    *   **실행:** Green 상태에서는 `transition`을 매우 느리게 설정하여, **안정된 상태가 지속됨**을 강조합니다. (긴급하지 않음)

현재까지 수집된 데이터(사용자의 '통제권 회복' 경험, '안전 기본값' 원칙)와 시장 트렌드를 바탕으로, 금융/관세 법규 변경에 따른 잠재적인 **Assumption Gap**을 분석하고 MVP의 차별화 포인트를 재설정할 방안을 제시합니다.

### 1. 핵심 Assumption Gap 분석 (Regulatory Friction)

현재 MVP는 주로 *운영 리스크*와 *심리적 불안정성* 관리에 초점을 맞추고 있습니다. 그러나 실제 비즈니스는 **규제 환경 변화**라는 거대한 외부 변수에 의해 직접적인 재무 및 운영 손실을 경험합니다. 이 간극이 현재의 MVP가 포착하지 못하는 가장 큰 Assumption Gap입니다.

| 발견된 Gap | 설명 (Assumption) | 현실적 리스크 (Reality Check) | 관련 감성 요소 |
| :--- | :--- | :--- | :--- |
| **Gap 1: 정적 규제 가정** | 관세/금융 법규는 비교적 정적이며 예측 가능하게 관리될 수 있다. | 급변하는 글로벌 공급망 및 각국의 환경, 무역 정책 변화에 따라 리스크가 실시간으로 변동하며, 과거 데이터의 유효성이 즉시 소멸된다. | 불안정성 (Anxiety) |
| **Gap 2: 단일 관점 분석** | 사용자는 재무적 손실(Margin Volatility)과 심리적 불안정성(Loss Aversion)을 분리하여 관리할 수 있다. | 규제 위반 또는 변경은 곧바로 재무적 손실로 직결되며, 이는 단순한 '불안함'이 아닌 **법적/재정적 책임**이라는 실존적 위험으로 전환된다. | 공포 (Fear) |
| **Gap 3: 플랫폼 의존도 가정** | 시스템의 안전 기본값(Safe Default Value)이 외부 환경 변화에도 일정 수준의 통제권을 제공한다. | 규제 변경에 따른 새로운 컴플라이언스 요구사항은 기존의 '안전 기본값'을 무효화시키거나, 새로운 책임 소재를 발생시켜 플랫폼 의존도가 오히려 위험 요소로 작용할 수 있다. | 불신 (Distrust) |

### 2. 새로운 불확실성 발굴 및 MVP 차별화 포인트 재설정

위의 Gap 분석을 바탕으로, MVP가 단순한 '안심 제공'을 넘어 **'규제 변화에 선제적으로 대응하는 통제권 확보 도구'**로 포지셔닝할 수 있는 새로운 불확실성 요소를 3가지 발굴하고 이를 차별화 포인트로 전환합니다.

#### A. 발굴된 새로운 불확실성 (New Uncertainties)

1.  **글로벌 공급망 리스크의 법적 귀속 문제 (Jurisdictional Risk):**
    *   **내용:** 특정 관세 또는 무역 규제가 변경될 때, 그 책임(Liability)이 수출입 과정의 어느 단계(원산지, 통관 대행사, 최종 수입자)에 귀속되는지에 대한 법적 해석이 국가별로 상이하며 예측 불가능하다.
    *   **MVP 적용 방향:** 단순한 비용 예측을 넘어, **'규제 변경 시 책임 분담 매트릭스(Liability Matrix)'**를 제공하여 잠재적 법적 리스크의 발생 지점을 시각화해야 합니다.

2.  **동적 컴플라이언스 요구사항 (Dynamic Compliance Demands):**
    *   **내용:** 환경 규제, 데이터 프라이버시(예: CBPR 등), 또는 특정 산업별 수출 제한 등이 실시간으로 업데이트되며, 이는 즉각적인 운영 비용과 프로세스 변경을 요구한다.
    *   **MVP 적용 방향:** 정적인 규칙 기반이 아닌, **'규제 변화 감지 및 자동 조정 모듈'**을 도입하여, 사용자가 새로운 규제가 발생했을 때 필요한 조치(Actionable Blueprint)를 즉시 제시해야 합니다.

3.  **AI/플랫폼의 법적 책임 경계 (Platform Accountability):**
    *   **내용:** 플랫폼이 제공하는 데이터 및 예측 모델에 대한 법적 책임 소재가 모호해지는 상황에서, 시스템이 제시한 '안전 기본값' 또는 '예측 결과'가 실제 재무적 손실 발생 시 어떤 수준의 책임을 지는지에 대한 불확실성이 증대된다.
    *   **MVP 적용 방향:** 시스템 내에 **'책임 경계 명시 프레임워크(Accountability Framework)'**를 통합하여, 플랫폼이 제공하는 정보의 한계와 사용자의 최종 의사결정 권한을 명확히 분리하고 시각화해야 합니다.

### 3. MVP 차별화 전략 재설정 (Strategic Pivot)

기존의 **'감성적 안정감 제공'**에서 **'선제적 규제 통제권 확보'**로 포지셔닝을 전환합니다.

| 기존 포지셔닝 | 재설정된 차별화 포인트 | 핵심 가치 제안 (Value Proposition) |
| :--- | :--- | :--- |
